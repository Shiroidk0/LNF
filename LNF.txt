    async function initFirebaseConnection(customConfig = null) {
      updateFirebaseStatusUI('connecting', 'Connecting Firebase...');

      try {
        // Tear down existing listeners & app instance if reconnecting
        if (unsubscribeItems) { unsubscribeItems(); unsubscribeItems = null; }
        if (unsubscribeClaims) { unsubscribeClaims(); unsubscribeClaims = null; }
        if (unsubscribeLogs) { unsubscribeLogs(); unsubscribeLogs = null; }

        const existingApps = getApps();
        for (const appInstance of existingApps) {
          await deleteApp(appInstance);
        }

        let configToUse = null;
        if (customConfig) {
          configToUse = customConfig;
          isCustomFirebase = true;
          activeAppId = customConfig.projectId || 'custom-campus-app';
        } else if (typeof __firebase_config !== 'undefined' && __firebase_config) {
          configToUse = JSON.parse(__firebase_config);
          isCustomFirebase = false;
          activeAppId = defaultAppId;
        } else {
          throw new Error('No Firebase configuration available.');
        }

        fbApp = initializeApp(configToUse);
        auth = getAuth(fbApp);
        db = getFirestore(fbApp);

        // RULE 3: Authenticate FIRST before attaching any Firestore queries
        if (!isCustomFirebase && typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }

        currentUser = auth.currentUser;
        if (currentUser) {
          const uidEl = document.getElementById('session-uid-display');
          if (uidEl) uidEl.textContent = currentUser.uid;
        }

        onAuthStateChanged(auth, (user) => {
          currentUser = user;
          const uidEl = document.getElementById('session-uid-display');
          if (uidEl && user) uidEl.textContent = user.uid;
        });

        updateFirebaseStatusUI(
          'connected',
          isCustomFirebase ? `Firebase (${configToUse.projectId})` : 'Firebase Live Sync'
        );

        attachFirestoreListeners();
      } catch (err) {
        console.error('Firebase initialization error:', err);
        updateFirebaseStatusUI('error', 'Firebase Offline');
        showToast('Firebase Connection Issue', err.message || 'Could not connect to Firestore.', 'error');
      }
    }

    function updateFirebaseStatusUI(status, label) {

            ? 'w-2.5 h-2.5 rounded-full bg-rose-500'

      }

      if (modalTitle && modalDesc) {

      }
    }

    /**
     * RULE 1 & RULE 2: Strict /artifacts/{appId}/public/data/{collectionName} paths
     * and simple collection() queries sorted in JS memory.
     */
    function getPublicCollectionRef(collectionName) {
      return collection(db, 'artifacts', activeAppId, 'public', 'data', collectionName);
    }

    function getPublicDocRef(collectionName, docId) {
      return doc(db, 'artifacts', activeAppId, 'public', 'data', collectionName, docId);
    }

    function attachFirestoreListeners() {
      if (!currentUser || !db) return;

      // 1. Items Collection Listener
      unsubscribeItems = onSnapshot(
        getPublicCollectionRef('items'),
        async (snapshot) => {
          const loadedItems = [];
          snapshot.forEach((docSnap) => {
            loadedItems.push({ id: docSnap.id, ...docSnap.data() });
          });

          // Sort newest first in memory
          loadedItems.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
          state.items = loadedItems;

          // Seed initial sample campus items automatically if database is empty on first load
          if (loadedItems.length === 0 && !state.hasSeededInitial) {
            state.hasSeededInitial = true;
            await seedDemoDataIfEmpty(false);
            return;
          }

          refreshAllUI();
        },
        (error) => {
          console.error('Firestore items listener error:', error);
          showToast('Firestore Sync Error', 'Unable to read items collection. Check Firestore rules.', 'error');
        }
      );

      // 2. Claims Collection Listener
      unsubscribeClaims = onSnapshot(
        getPublicCollectionRef('claims'),
        (snapshot) => {
          const loadedClaims = [];
          snapshot.forEach((docSnap) => {
            loadedClaims.push({ claimId: docSnap.id, ...docSnap.data() });
          });
          loadedClaims.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
          state.claims = loadedClaims;
          refreshAllUI();
        },
        (error) => {
          console.error('Firestore claims listener error:', error);
        }
      );

      // 3. Notifications / Audit Logs Listener
      unsubscribeLogs = onSnapshot(
        getPublicCollectionRef('notifications'),
        (snapshot) => {
          const loadedLogs = [];
          snapshot.forEach((docSnap) => {
            loadedLogs.push({ id: docSnap.id, ...docSnap.data() });
          });
          loadedLogs.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
          state.logs = loadedLogs;
          renderActivityLogs();
        },
        (error) => {
          console.error('Firestore notifications listener error:', error);
        }
      );
    }

    async function writeFirestoreNotification(type, title, detail) {
      if (!currentUser || !db) return;
      const now = new Date();
      const timestamp =
        now.toLocaleDateString('en-US', { month: 'short', day: 'numeric' }) +
        ', ' +
        now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
      const logId = 'LOG-' + Date.now() + '-' + Math.floor(Math.random() * 100);

      await setDoc(getPublicDocRef('notifications', logId), {
        id: logId,
        timestamp,
        type,
        title,
        detail,
        createdAt: Date.now()
      });
    }

        const matchesCategory = state.selectedCategory === 'All' || item.category === state.selectedCategory;
        const matchesStatus = state.selectedStatus === 'All' || item.status === state.selectedStatus;
        const matchesSearch =
          !state.searchQuery ||
          item.title.toLowerCase().includes(state.searchQuery) ||
          item.description.toLowerCase().includes(state.searchQuery) ||
          item.location.toLowerCase().includes(state.searchQuery) ||
          item.id.toLowerCase().includes(state.searchQuery);

    function handlePhotoUpload(event) {
      const file = event.target.files[0];
      if (!file) return;

      if (!file.type.startsWith('image/')) {
        showToast('Invalid File', 'Please select a valid image file (PNG, JPG, WEBP).', 'error');
        return;
      }

      const reader = new FileReader();
      reader.onload = function (e) {
        const img = new Image();
        img.onload = function () {
          // Compress image via HTML5 Canvas so it easily fits within Firestore 1MB document limit
          const canvas = document.createElement('canvas');
          const MAX_WIDTH = 640;
          const scale = Math.min(1, MAX_WIDTH / img.width);
          canvas.width = img.width * scale;
          canvas.height = img.height * scale;
          const ctx = canvas.getContext('2d');
          ctx.drawImage(img, 0, 0, canvas.width, canvas.height);

          state.uploadedPhotoDataUrl = canvas.toDataURL('image/jpeg', 0.75);
          const previewImg = document.getElementById('report-photo-preview');
          const placeholder = document.getElementById('report-photo-placeholder');
          previewImg.src = state.uploadedPhotoDataUrl;
          previewImg.classList.remove('hidden');
          placeholder.classList.add('hidden');
          showToast('Photo Ready', 'Image compressed and attached for Firestore upload.', 'info');
        };
        img.src = e.target.result;
      };
      reader.readAsDataURL(file);
    }

    async function handleClaimSubmit(event) {
      event.preventDefault();
      if (!currentUser || !db) return;

      const itemId = document.getElementById('claim-item-id').value;
      const claimantName = document.getElementById('claimant-name').value.trim();
      const claimantId = document.getElementById('claimant-id').value.trim();
      const claimantEmail = document.getElementById('claimant-email').value.trim();
      const proof = document.getElementById('claim-proof').value.trim();

      const item = state.items.find(i => i.id === itemId);
      if (!item) return;

      const claimId = 'CLM-' + Math.floor(100 + Math.random() * 899);
      const nowStr = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

      try {
        // 1. Write claim verification document to Firestore
        await setDoc(getPublicDocRef('claims', claimId), {
          claimId,
          itemId: item.id,
          itemTitle: item.title,
          claimantName,
          claimantId,
          claimantEmail,
          claimantUid: currentUser.uid,
          proof,
          status: 'Pending',
          submittedAt: `Today, ${nowStr}`,
          createdAt: Date.now()
        });

        // 2. Automatically update item status in Firestore to 'Claim Pending'
        await updateDoc(getPublicDocRef('items', item.id), {
          status: 'Claim Pending',
          activeClaimId: claimId
        });

        // 3. Log automated notification event
        await writeFirestoreNotification(
          'CLAIM_SUBMITTED',
          `Automated Hold: ${item.id} (${item.title})`,
          `Status auto-updated Available → Claim Pending. Email/SMS alerts sent to claimant (${claimantEmail}) & finder (${item.finderEmail}).`
        );

        closeClaimModal();
        showToast(
          'Claim Submitted to Firebase',
          `Status updated to "Claim Pending". Automated alerts dispatched to you and ${item.finderName}.`,
          'warning'
        );
      } catch (err) {
        console.error('Claim submission error:', err);
        showToast('Claim Error', err.message || 'Could not submit claim to Firestore.', 'error');
      }
    }

    async function processClaimDecision(claimId, itemId, decision) {
      if (!currentUser || !db) return;
      const claim = state.claims.find(c => c.claimId === claimId);
      const item = state.items.find(i => i.id === itemId);
      if (!claim) return;

      try {
        if (decision === 'approve') {
          await updateDoc(getPublicDocRef('claims', claimId), { status: 'Approved' });
          if (item) {
            await updateDoc(getPublicDocRef('items', itemId), { status: 'Returned' });
          }
          await writeFirestoreNotification(
            'CLAIM_APPROVED',
            `Claim Approved (${claimId}): ${itemId} → Returned`,
            `Ownership verified for ${claim.claimantName}. Automated pickup pass sent to ${claim.claimantEmail}.`
          );
          showToast('Claim Approved → Status: Returned', `Firestore updated & notifications sent to ${claim.claimantName}.`, 'success');
        } else {
          await updateDoc(getPublicDocRef('claims', claimId), { status: 'Rejected' });
          if (item) {
            await updateDoc(getPublicDocRef('items', itemId), { status: 'Available', activeClaimId: null });
          }
          await writeFirestoreNotification(
            'CLAIM_REJECTED',
            `Claim Rejected (${claimId}): ${itemId} → Available`,
            `Verification details did not match. Item automatically re-listed as Available. Notification sent to ${claim.claimantEmail}.`
          );
          showToast('Claim Rejected → Item Re-listed', `Item ${itemId} restored to Available in Firestore.`, 'info');
        }
      } catch (err) {
        console.error('Decision error:', err);
        showToast('Update Error', err.message || 'Could not update Firestore record.', 'error');
      }
    }

    function showToast(title, message, variant = 'info') {
      const container = document.getElementById('toast-container');
      const toast = document.createElement('div');

      const styles = {
        success: { border: 'border-emerald-200', iconBg: 'bg-emerald-100 text-emerald-600', icon: 'fa-circle-check' },
        warning: { border: 'border-amber-200', iconBg: 'bg-amber-100 text-amber-600', icon: 'fa-bolt' },
        error: { border: 'border-rose-200', iconBg: 'bg-rose-100 text-rose-600', icon: 'fa-circle-exclamation' },
        info: { border: 'border-campus-200', iconBg: 'bg-campus-100 text-campus-600', icon: 'fa-bell' }
      }[variant] || { border: 'border-slate-200', iconBg: 'bg-slate-100 text-slate-600', icon: 'fa-info-circle' };

      toast.className = `toast-enter pointer-events-auto bg-white rounded-xl shadow-lg border ${styles.border} p-3.5 flex items-start gap-3`;
      toast.innerHTML = `
        <div class="w-8 h-8 rounded-lg ${styles.iconBg} flex items-center justify-center shrink-0 mt-0.5">
          <i class="fa-solid ${styles.icon}"></i>
        </div>
        <div class="flex-1 min-w-0">
          <p class="text-xs font-bold text-slate-900">${escapeHTML(title)}</p>
          <p class="text-xs text-slate-600 mt-0.5 leading-snug">${escapeHTML(message)}</p>
        </div>
        <button onclick="this.parentElement.remove()" class="text-slate-400 hover:text-slate-600 p-1">
          <i class="fa-solid fa-xmark text-xs"></i>
        </button>
      `;

      container.appendChild(toast);

      setTimeout(() => {
        if (toast.parentElement) {
          toast.style.opacity = '0';
          toast.style.transform = 'translateX(100%)';
          toast.style.transition = 'all 0.25s ease';
          setTimeout(() => toast.remove(), 250);
        }
      }, 4800);
    }

    function escapeHTML(str) {
      if (!str) return '';
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }
